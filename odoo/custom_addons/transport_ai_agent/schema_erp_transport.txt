=== RÈGLES GÉNÉRALES ===
- Base PostgreSQL, encodage UTF-8
- Les champs name de type jsonb : utiliser name->>'fr_FR' ou name->>'en_US'
- Ne jamais utiliser "name" brut pour les tables avec name jsonb
- Ne jamais utiliser "name" pour transport_assurance_bus → utiliser "numero_police"
- Les champs Many2one stockent un INTEGER (foreign key)
- Toujours utiliser ILIKE pour les recherches texte (insensible à la casse)
- LIMIT 20 par défaut sauf COUNT

=== JOINTURES FRÉQUENTES — CORRIGÉES ===

-- Tournée avec bus, chauffeur et ligne (name jsonb)
SELECT t.name, t.state, t.direction,
       COALESCE(l.name->>'fr_FR', l.name->>'en_US') AS ligne,
       v.name AS bus, v.license_plate,
       e.name AS chauffeur
FROM transport_exploitation_tournee t
LEFT JOIN transport_exploitation_ligne l ON t.ligne_id = l.id
LEFT JOIN fleet_vehicle v ON t.vehicle_id = v.id
LEFT JOIN hr_employee e ON t.chauffeur_id = e.id

-- Police assurance avec compagnie, type (jsonb) et bus
SELECT a.numero_police, a.state, a.date_debut, a.date_fin,
       COALESCE(tp.name->>'en_US', tp.name->>'fr_FR') AS type_police,
       c.name AS compagnie, v.name AS bus
FROM transport_assurance_bus a
LEFT JOIN fleet_vehicle v ON a.vehicle_id = v.id
LEFT JOIN transport_assurance_compagnie c ON a.compagnie_id = c.id
LEFT JOIN transport_assurance_type tp ON a.type_police_id = tp.id

-- Bon carburant avec bus (via fuel_voucher_line !)
SELECT f.name, f.voucher_type, f.total_quantity, f.state, f.date,
       v.name AS bus, v.license_plate
FROM transport_fuel_voucher f
JOIN transport_fuel_voucher_line fl ON fl.voucher_id = f.id
JOIN fleet_vehicle v ON fl.vehicle_id = v.id

-- Sinistre avec bus et chauffeur (employe_id pas chauffeur_id)
SELECT s.name, s.state, s.nature_sinistre,
       s.montant_accorde, s.montant_net_verse,
       v.name AS bus, e.name AS chauffeur
FROM transport_assurance_sinistre s
LEFT JOIN fleet_vehicle v ON s.vehicle_id = v.id
LEFT JOIN hr_employee e ON s.employe_id = e.id

-- Courrier avec expéditeur (jsonb)
SELECT c.name, c.sujet,
       COALESCE(c.expediteur_nom->>'fr_FR', c.expediteur_nom->>'en_US') AS expediteur,
       c.state, c.date_arrivee, c.en_retard
FROM boc_courrier_arrivee c
